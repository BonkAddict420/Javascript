                            function f_roh(x_roh) {
                                if (x_roh >= 2) {
                                    return 9;
                                } else if (x_roh >= 0 && x_roh < 2) {
                                    return Math.pow(x_roh, 3) + 1;
                                } else if (x_roh > -2 && x_roh < 0) {
                                    return Math.pow(x_roh + 1, 2);
                                } else if (x_roh <= -2) {
                                    return 1;
                                }
                            }

                            document.write("f(" + x_roh.toFixed(2) + ") = " + f_roh(x_roh).toFixed(2) + "<br>");
