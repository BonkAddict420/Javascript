    <script>
        function Pole_kowalski(a_kow, b_kow) {
            var pole = parseFloat(a_kow) * parseFloat(b_kow);
            document.getElementById("wyprowadz_pole_kow").value = pole.toFixed(2);
        }

        function obwod_kowalski(a_kow, b_kow) {
            var obw = 2 * parseFloat(a_kow) + 2 * parseFloat(b_kow);
            document.getElementById("wyprowadz_obwod_kow").value = obw.toFixed(2);
        }
    </script>
