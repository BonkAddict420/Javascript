    <script type="text/javascript">
        function pole_roh() {
            var a_roh = parseFloat(document.getElementById('dlugosc').value);
            var b_roh = a_roh * a_roh;
            document.getElementById("nazwa").innerHTML = "Pole kwadratu= " + b_roh.toFixed(2) + " cm^2";
        }
        function obwod_roh() {
            var a_roh = parseFloat(document.getElementById('dlugosc').value);
            var b_roh = 4 * a_roh;
            document.getElementById("nazwa").innerHTML = "Obwód kwadratu= " + b_roh.toFixed(2) + " cm";
        } 
    </script>
